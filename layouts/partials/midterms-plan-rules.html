{{- /* What the voting plan offers, for one state. Takes (dict "state" VOTING_ENTRY "notes" PLAN.notes) and returns a dict. First, one entry per kind of date the plan asks for: early the day you vote early: the early voting window, without Election Day, which the plan asks about on its own fill the day you fill out a mail ballot return the day you send it back, before you've said how return_mail ... by mail return_dropbox ... at a drop box return_office ... in person Each is {ranges: [{from, to}, ...], note}, and return_mail adds late_from, the first day of the last week before the deadline, by, the week-early date we recommend, and deadline, the state's rule ("arrive by Tue, Nov 3 at 8PM"). A range is two ISO dates, and an empty `from` means "today", which only the reader's browser knows; the picker clips every range to today as well. Election Day is the fixed last day. Then what the rest of the form needs from the state: ways which of the plan's ways (person, mail) the state has choices for a choice blank with a `rule`, the option keys the state allows (a state with no early voting allows only Election Day, so the form picks it and skips the question) hints for a step with a `hint`, the line shown under it warn for a step with a `state_warn`, the warning shown under it id_url the state's ID page from week three, linked after any line above that is about ID: the ID hint, a hand-in note marked id_link, and an envelope warning named in id_warn Dates come from data/midterms_voting.yml, so a state's picker follows the same dates its week-two card prints, with the exceptions in data/midterms_return.yml. ID needs come from week three's data/midterms_id.yml. */ -}} {{- $s := .state -}} {{- $notes := .notes -}} {{- $extra := site.Data.midterms_return -}} {{- $idData := index site.Data.midterms_id.states $s.code -}} {{- $election := "2026-11-03" -}} {{- /* Early voting: the state's window, or for a state whose counties set their own, any day up to its last. The line under the early-or-Election- Day question gives the dates; a state without them votes on the day. */ -}} {{- $early := slice -}} {{- $earlyHint := "" -}} {{- $electionHint := "" -}} {{- if $s.early_voting -}} {{- if $s.ev_varies -}} {{- $end := "2026-11-02" -}} {{- with $s.ev_end }}{{ $end = partial "midterms-iso.html" . }}{{ end -}} {{- $early = $early | append (slice (dict "from" "" "to" $end)) -}} {{- $earlyHint = $notes.county -}} {{- else if and $s.ev_start $s.ev_end -}} {{- $early = $early | append (slice (dict "from" (partial "midterms-iso.html" $s.ev_start) "to" (partial "midterms-iso.html" $s.ev_end))) -}} {{- $earlyHint = replace (replace (replace $notes.early_dates "{state}" $s.name) "{from}" $s.ev_start) "{to}" $s.ev_end -}} {{- end -}} {{- end -}} {{- $voteWhen := slice "election" -}} {{- if $early }}{{ $voteWhen = slice "early" "election" }}{{ else }}{{ $electionHint = replace $notes.no_early "{state}" $s.name }}{{ end -}} {{- /* By mail: until the mail deadline. We ask people to return a ballot a week early, so the picker shades the last seven days ending on it. return_mail_by is that same week-early date; late_from is the day after. */ -}} {{- $mailLast := partial "midterms-iso.html" $s.return_mail -}} {{- $mailDeadline := time.AsTime (printf "%sT12:00:00Z" $mailLast) -}} {{- $mailNote := "" -}} {{- /* The mail deadline as the late warning states it: arrive by, or be postmarked by, the day and time. */ -}} {{- $mailWhen := $s.return_mail -}} {{- with $s.return_mail_time }}{{ $mailWhen = printf "%s at %s" $mailWhen . }}{{ end -}} {{- $mailRule := cond (eq $s.return_mail_received true) $notes.deadline_received $notes.deadline_postmarked -}} {{- with $s.return_mail_by }}{{ $mailNote = replace $notes.mail_by "{date}" . }}{{ end -}} {{- $returnMail := dict "ranges" (slice (dict "from" "" "to" $mailLast)) "note" $mailNote "late_from" ($mailDeadline.AddDate 0 0 -6 | time.Format "2006-01-02") "by" $s.return_mail_by "deadline" (replace $mailRule "{date}" $mailWhen) -}} {{- $ways := slice "mail" -}} {{- $returns := dict "mail" $returnMail -}} {{- $last := $mailLast -}} {{- /* In person: until the in-person deadline, in states that take a ballot by hand at all. Handing it in has no transit time, so there's no late week, just the deadline. */ -}} {{- with $s.return_in_person -}} {{- $to := partial "midterms-iso.html" . -}} {{- $deadline := . -}} {{- with $s.return_in_person_time }}{{ $deadline = printf "%s at %s" $deadline . }}{{ end -}} {{- $note := replace (replace $notes.in_person "{state}" $s.name) "{deadline}" $deadline -}} {{- $idNote := in $extra.in_person_id $s.code -}} {{- if $idNote }}{{ $note = printf "%s %s" $note (replace $notes.in_person_id "{state}" $s.name) }}{{ end -}} {{- $returns = merge $returns (dict "office" (dict "ranges" (slice (dict "from" "" "to" $to)) "note" $note "id_link" $idNote)) -}} {{- $ways = $ways | append "office" -}} {{- if gt $to $last }}{{ $last = $to }}{{ end -}} {{- end -}} {{- /* At a drop box: wherever the state lists one, on the in-person days unless midterms_return.yml says otherwise. */ -}} {{- if in (lower (string $s.return_methods)) "drop box" -}} {{- $o := index $extra.drop_box $s.code | default dict -}} {{- $date := $s.return_mail -}} {{- $time := "" -}} {{- with $s.return_in_person }}{{ $date = . }}{{ $time = $s.return_in_person_time | default "" }}{{ end -}} {{- with $o.to }}{{ $date = . }}{{ $time = $o.time | default "" }}{{ end -}} {{- $from := "" -}} {{- with $o.from }}{{ $from = partial "midterms-iso.html" . }}{{ end -}} {{- $to := partial "midterms-iso.html" $date -}} {{- $deadline := $date -}} {{- with $time }}{{ $deadline = printf "%s at %s" $deadline . }}{{ end -}} {{- $note := replace (replace $notes.drop_box "{state}" $s.name) "{deadline}" $deadline -}} {{- $returns = merge $returns (dict "dropbox" (dict "ranges" (slice (dict "from" $from "to" $to)) "note" $note)) -}} {{- $ways = $ways | append "dropbox" -}} {{- if gt $to $last }}{{ $last = $to }}{{ end -}} {{- end -}} {{- /* Before a way back is picked, any day one of them allows. */ -}} {{- $window := slice (dict "from" "" "to" $last) -}} {{- /* ID at the polls. Buckets A-D ask voters for ID, so the plan asks whether the reader has it, with the state's line under the question. Bucket E asks no one but first-time voters, so it allows only "none" and the question is skipped. The keys are the ID blank's option keys in data/midterms.yml. */ -}} {{- $id := slice "none" -}} {{- $idHint := "" -}} {{- with $idData -}} {{- if ne .bucket "E" }}{{ $id = slice "yes" "no" }}{{ end -}} {{- $idHint = printf "%s." .lede -}} {{- end -}} {{- $envelope := index $extra.envelope $s.code | default "" -}} {{- $planWays := cond (in $extra.mail_only $s.code) (slice "mail") (slice "person" "mail") -}} {{- return dict "early" (dict "ranges" $early "note" "") "fill" (dict "ranges" $window "note" "") "return" (dict "ranges" $window "note" "") "return_mail" $returnMail "return_office" (index $returns "office") "return_dropbox" (index $returns "dropbox") "ways" $planWays "choices" (dict "return_how" $ways "id" $id "vote_when" $voteWhen) "hints" (dict "id" $idHint "vote_when" $earlyHint "election" $electionHint) "warn" (dict "envelope" $envelope) "id_warn" (dict "envelope" (findRE `\bID\b` $envelope | len | lt 0)) "id_url" (cond (isset $idData "url") (index $idData "url") "") -}}
