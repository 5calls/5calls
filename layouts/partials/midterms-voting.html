{{- /* One state's voting options for the week-two card, as HTML. Takes (dict "state" STATE "part" "lede"|"body") and returns a string, the same contract as midterms-deck.html: the lede sits beside the state tag, the body under it. Returning a string lets the page print the reader's state and the state switcher carry every other state's, drawn by the same code. The state comes from data/midterms_voting.yml, generated by scripts/build-midterms-voting.py. Every option is shown by asking whether its key is there — a state without early voting, without same-day registration or without an online ballot request simply has no key, and the line is not drawn. Four groups, each a lead line and an optional list under it: In-person early voting the window, or a note that the county sets it Same-day registration when you can do it, and where Absentee & vote-by-mail what the state offers, then how to ask for one Returning your ballot where it can go, then by when A deadline may carry a time it falls at, whether the date is when the ballot must arrive or be postmarked, and a note qualifying it. Where upstream gives prose instead of a date — Mississippi and Tennessee cannot hand-deliver a ballot at all — the prose is the line. */ -}} {{- $s := .state -}} {{- $out := "" -}} {{- if eq .part "lede" -}} {{- $out = printf `

Voting options in %s:

` $s.name -}} {{- else -}} {{- $groups := slice -}} {{- /* A date, with the time it falls at where there is one: "Mon, Nov 2 at 12PM". Takes the state and the key's prefix. */ -}} {{- $when := dict -}} {{- /* In-person early voting. Six states leave the dates to the county, so the option is real but the window is not statewide. */ -}} {{- if $s.early_voting -}} {{- $line := "" -}} {{- if $s.ev_varies -}} {{- $line = "Yes — your county sets the dates" -}} {{- with $s.ev_end -}} {{- $line = printf "%s, through %s" $line . -}} {{- with $s.ev_end_time }}{{ $line = printf "%s at %s" $line . }}{{ end -}} {{- end -}} {{- else -}} {{- $start := $s.ev_start -}} {{- $end := $s.ev_end -}} {{- with $s.ev_end_time }}{{ $end = printf "%s at %s" $end . }}{{ end -}} {{- $line = printf "Yes, from %s to %s" $start $end -}} {{- end -}} {{- $groups = $groups | append (dict "label" "In-person early voting" "text" $line) -}} {{- end -}} {{- /* Same-day registration. The two windows are independent: New Hampshire registers on Election Day but has no early voting period at all. */ -}} {{- if $s.same_day_reg -}} {{- $windows := slice -}} {{- if $s.sdr_early_voting }}{{ $windows = $windows | append "during early voting" }}{{ end -}} {{- /* Election Day is the one fixed date on this page. */ -}} {{- if $s.sdr_election_day }}{{ $windows = $windows | append "on Election Day, Nov 3" }}{{ end -}} {{- $line := "Yes" -}} {{- with $windows }}{{ $line = printf "Yes, %s" (delimit . " and ") }}{{ end -}} {{- $sub := slice -}} {{- with $s.sdr_locations }}{{ $sub = $sub | append (markdownify .) }}{{ end -}} {{- $groups = $groups | append (dict "label" "Same-day registration" "text" $line "sub" $sub) -}} {{- end -}} {{- /* Asking for a mail ballot. The automatic-ballot states have no request deadlines at all, so the list under the lead line is empty and only what the state offers is named. */ -}} {{- with $s.absentee_type -}} {{- $sub := slice -}} {{- with $s.request_online -}} {{- $date := . -}} {{- with $s.request_online_time }}{{ $date = printf "%s at %s" $date . }}{{ end -}} {{- $sub = $sub | append (printf "Online: Request by %s" $date) -}} {{- end -}} {{- with $s.request_online_note -}} {{- $sub = $sub | append (printf "Online: %s" (markdownify .)) -}} {{- end -}} {{- with $s.request_mail -}} {{- $by := cond (eq $s.request_mail_received false) "postmarked" "received" -}} {{- $date := . -}} {{- with $s.request_mail_time }}{{ $date = printf "%s at %s" $date . }}{{ end -}} {{- $sub = $sub | append (printf "By mail: Request must be %s by %s" $by $date) -}} {{- end -}} {{- with $s.request_mail_note -}} {{- $sub = $sub | append (printf "By mail: %s" (markdownify .)) -}} {{- end -}} {{- with $s.request_in_person -}} {{- $date := . -}} {{- with $s.request_in_person_time }}{{ $date = printf "%s at %s" $date . }}{{ end -}} {{- $sub = $sub | append (printf "In person: Request by %s" $date) -}} {{- end -}} {{- with $s.request_in_person_note -}} {{- $sub = $sub | append (printf "In person: %s" (markdownify .)) -}} {{- end -}} {{- $groups = $groups | append (dict "label" "Absentee & vote-by-mail" "text" . "sub" $sub) -}} {{- end -}} {{- /* Sending it back. Two states have no places listed upstream, so the lead line names only the post. */ -}} {{- $return := slice -}} {{- with $s.return_mail -}} {{- $by := cond (eq $s.return_mail_received false) "Postmarked" "Received" -}} {{- $date := . -}} {{- with $s.return_mail_time }}{{ $date = printf "%s at %s" $date . }}{{ end -}} {{- $line := printf "By mail: %s by %s" $by $date -}} {{- /* Where the postmark is what counts, the date it must actually turn up matters just as much: without it the line reads as though arriving late is fatal, which in these states it is not. */ -}} {{- with $s.return_mail_received_by }}{{ $line = printf "%s, and must arrive by %s" $line . }}{{ end -}} {{- with $s.return_mail_by }}{{ $line = printf "%s (mail it by %s!)" $line . }}{{ end -}} {{- $return = $return | append $line -}} {{- end -}} {{- with $s.return_in_person -}} {{- $date := . -}} {{- with $s.return_in_person_time }}{{ $date = printf "%s at %s" $date . }}{{ end -}} {{- $return = $return | append (printf "In person: Received by %s" $date) -}} {{- end -}} {{- with $s.return_in_person_note -}} {{- $return = $return | append (printf "In person: %s" (markdownify .)) -}} {{- end -}} {{- if $return -}} {{- $where := "By mail" -}} {{- with $s.return_methods }}{{ $where = printf "%s, or by mail" . }}{{ end -}} {{- $groups = $groups | append (dict "label" "How to return your ballot" "text" $where "sub" $return) -}} {{- end -}} {{- range $groups -}} {{- $items := "" -}} {{- range .sub }}{{ $items = printf "%s
  • %s
    • " $items . }}{{ end -}} {{- $list := "" -}} {{- with $items }}{{ $list = printf `
      %s
    ` . }}{{ end -}} {{- $out = printf `%s

    %s: %s

    %s
    ` $out .label .text $list -}} {{- end -}} {{- $out = printf `
    %s
    ` $out -}} {{- end -}} {{- return $out -}}
